# SIMD Kernel Manifest (PLAN-ASSEMBLY-REWRITE Section 2)
#
# Single source of truth consumed by build.zig at configure time:
#   - Drives NASM compile steps for every kernel listed.
#   - Generates `simd_kernels_manifest.h` in the build cache, which is included
#     by `src/simd/ggml_simd_hook.cpp` instead of hand-maintained
#     `extern void simd_*` declarations.
#
# Format (whitespace-separated; '#' starts a comment):
#   <arch>  <relative_path>                       <symbol>                       <signature>
#
# Signatures (C-side prototypes emitted into the generated header):
#   vec_dot       void(int n, float* r, const void* vx, const void* vy);
#   rms_norm_f32  void(int n, float eps, const float* x, float* y);
#   rope_neox_f32 void(long long n_pairs, const float* cache, const float* src, float* dst);
#
# Architectures: x86 (built on x86_64 hosts), arm64 (built on aarch64 hosts).
# The build adds NASM steps for `x86` rows and GAS .S sources for `arm64` rows.
